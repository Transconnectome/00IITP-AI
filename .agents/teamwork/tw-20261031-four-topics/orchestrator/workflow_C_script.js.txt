async function loop(mk) {
  const m = MS[mk]
  const history = []
  for (let k = 0; k < MAX_ITER; k++) {
    const it = m.startIt + k
    const last = history.length ? history[history.length - 1] : null
    const prior = last ? `\n\nPREVIOUS ITERATION FAILURE (gate record incl. every verifier/victory report verbatim — address every blocking issue; fix integrity issues, never circumvent):\n${JSON.stringify(last)}` : (m.prior ? `\n\nPREVIOUS ITERATION FAILURE is recorded in ${m.prior} (read it fully; address every blocking issue).` : '')
    const lenses = ['scientific design and falsifiability (card, controls, kill criterion, permitted data)', 'feasibility and failure modes (code reality, Slurm/data logistics, cost measured by cheap probes, fixtures that must fail on the targeted defect)', 'adversarial pre-mortem (how could this milestone produce a wrong or unverifiable verdict; integrity and C4/C6 hazards; what verifiers will attack)']
    const exps = (await parallel([1, 2, 3].map(e => () => agent(COMMON(`explorer_${mk}_r${it}_${e}`, 'explorer') + `

Task: milestone ${mk}: ${m.title}. ${last || m.prior ? 'Analyze the previous failure and recommend a fix strategy. ' : 'Give the worker a concrete, verifiable strategy covering every criterion, with evidence from cheap probes (synthetic data, read-only checks, at most a 0.25 GPU-h ledgered Slurm smoke test). '}Explorer ${e} of 3 — primary lens: ${lenses[e - 1]}. Recommend; do NOT implement production code. Work independently.
Milestone order of work: ${m.order}${prior}`,
      { label: `explorer_${mk}_r${it}_${e}`, phase: 'Explore', schema: EXPLORER_SCHEMA })))).filter(Boolean)
    log(`${mk} iter ${it}: ${exps.length}/3 explorers`)
    const w = await agent(COMMON(`worker_${mk}_r${it}`, 'worker') + `

Task: implement milestone ${mk}: ${m.title}, satisfying ACCEPTANCE C1–C8 and the milestone criteria. Exclusive write ownership: ${m.own} ${k > 0 ? 'Previous worker output exists in these paths; you own them now. Preserve prior bytes of anything you change in ' + P + '/verification/worker_' + mk + '_r' + it + '/prev_snapshot/. Frozen cards are never edited; deviations go in a dated addendum.' : ''}
${m.order}
Run all relevant tests (new + old 86 via: ${OLD_SUITE}) and the protected hash check before returning.

DO NOT CHEAT. All implementations must be genuine. DO NOT hardcode test results, create dummy/facade implementations, or circumvent the intended task. A teamwork_preview_auditor will independently verify your work. Integrity violations WILL be detected and your work WILL be rejected.

Explorer strategies (iteration ${it}): ${JSON.stringify(exps)}${prior}`, { label: `worker_${mk}_r${it}`, phase: 'Implement', schema: WORKER_SCHEMA })
    if (!w) { history.push({ milestone: mk, iteration: it, worker: null, gate: 'FAIL (worker null)' }); log(`${mk} iter ${it}: worker null`); continue }
    const art = `Milestone ${mk}: ${m.title}. Raw artifacts are the paths in the worker ownership: ${m.own}. ${m.verify} Worker's structured handoff (claims to verify, not trust): ${JSON.stringify(w)}`
    const instr = {
      reviewer: `Independently review correctness and completeness vs ACCEPTANCE C1–C8 and the milestone criteria; re-run tests and the protected hash check yourself. APPROVE only if every criterion is met with evidence (or an untestable/escape outcome is properly evidenced and costed); else REQUEST_CHANGES with blocking_issues.`,
      challenger: `Empirically try to break it: independent re-implementations of key computations, adversarial synthetic inputs, mutation tests that each fixture/verdict rule must catch, recomputation of headline numbers from saved raw outputs, checks that the card was frozen before results (timestamps + hashes) and that the verdict follows the card's rules. CONFIRMED if it holds, REFUTED with evidence.`,
      auditor: `Forensic integrity audit per your role: hardcoded results, facades, tests that cannot fail, fabricated/non-reproducible logs, results computed before card freeze, silent fallbacks, participant-ID leakage (incl. recoverable hashes), writes to protected/pinned/external paths, GPU use outside Slurm, not in the ledger or beyond cap, another project's env, git writes by agents. Treat heuristics as leads needing corroboration. CLEAN or INTEGRITY_VIOLATION with evidence.`,
    }
    const roles = [['reviewer', 1], ['reviewer', 2], ['challenger', 1], ['challenger', 2], ['auditor', 1]]
    const got = (await parallel(roles.map(([r, n]) => () => {
      const id = `${r}_${mk}_r${it}_${n}`
      return agent(COMMON(id, r) + `\n\nFresh independent context, no implementation history. Read-only on production files. ${instr[r]}\n\n${art}`,
        { label: id, phase: 'Verify', schema: VERDICT_SCHEMA }).then(v => ({ id, role: r, v }))
    }))).filter(Boolean)
    const by = r => got.filter(x => x.role === r)
    const reasons = []
    const aud = by('auditor')[0]
    if (!aud || !aud.v) reasons.push('auditor missing'); else if (aud.v.verdict !== 'CLEAN') reasons.push('auditor ' + aud.v.verdict)
    if (w.status !== 'DONE') reasons.push('worker ' + w.status)
    const revs = by('reviewer'), chs = by('challenger')
    if (revs.length < 2 || revs.some(x => !x.v)) reasons.push('reviewer missing')
    revs.forEach(x => { if (x.v && x.v.verdict !== 'APPROVE') reasons.push(x.id + ' ' + x.v.verdict) })
    if (chs.length < 2 || chs.some(x => !x.v)) reasons.push('challenger missing')
    chs.forEach(x => { if (x.v && x.v.verdict !== 'CONFIRMED') reasons.push(x.id + ' ' + x.v.verdict) })
    const pass = reasons.length === 0
    const rec = { milestone: mk, iteration: it, explorers: exps, worker: w, verifiers: got, gate: pass ? 'PASS' : 'FAIL (' + reasons.join('; ') + ')' }
    history.push(rec)
    log(`${mk} iter ${it} gate: ${rec.gate}`)
    if (pass) {
      const va = await agent(COMMON(`victory_${mk}_r${it}`, 'victory-auditor') + `

You have ZERO shared context with the implementation team. Verify the claimed completion of milestone ${mk}: ${m.title} against ORIGINAL_REQUEST.md, ACCEPTANCE.md (C1–C8 + this milestone's criteria) and PI decisions 1–8 yourself from raw artifacts (${m.own}). Do not read team handoff.md files before forming your verdict. ${m.verify} Return verdict VERIFIED_COMPLETE or REJECTED with the full VICTORY AUDIT REPORT in full_report, and list in nonblocking_issues what the human must know.`,
        { label: `victory_${mk}_r${it}`, phase: 'Victory', schema: VERDICT_SCHEMA })
      rec.victory = va
      log(`${mk} victory: ${va ? va.verdict : 'null'}`)
      if (va && va.verdict === 'VERIFIED_COMPLETE') break
      rec.gate = 'PASS but victory ' + (va ? va.verdict : 'null')
    }
  }
  const fin = history.length ? history[history.length - 1] : null
  const status = fin && fin.victory && fin.victory.verdict === 'VERIFIED_COMPLETE' ? 'VERIFIED_COMPLETE' : 'INCOMPLETE'
  if (status !== 'VERIFIED_COMPLETE') log(`${mk}: ended INCOMPLETE after ${history.length} iteration(s) — PI decision needed`)
  return { milestone: mk, status, history }
}
